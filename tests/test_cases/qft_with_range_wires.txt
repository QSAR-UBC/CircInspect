from pennylane import numpy as np
import pennylane as qp
def qfunc_adder(m, wires):
    qp.QFT(wires=wires)
    for i in range(len(wires)): 
        const = m*np.pi/(2**i)
        qp.PhaseShift(const ,wires=[i])   
    qp.adjoint(qp.QFT(wires=wires))      
wires = range(4) 
dev = qp.device("default.qubit", wires=wires, shots=1) 
@qp.qnode(dev)
def test_circuit():
    qp.PauliX(wires=0)
    qfunc_adder(5, wires)
    return qp.sample()
test_circuit()
