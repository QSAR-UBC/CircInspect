import pennylane as qp
from pennylane import numpy as np

dev = qp.device("default.qubit", wires=2)

def hello():
    qp.RZ(1,1)

@qp.qnode(dev)
def circuit1(x):
    qp.RX(x,0)
    hello()
    qp.CNOT([0,1])
    return qp.probs()

@qp.qnode(dev) 
def circuit2(x):
    qp.RZ(x,0)
    hello()
    qp.CNOT([0,1])
    return qp.probs()

circuit1(0.1)
circuit2(0.2)
