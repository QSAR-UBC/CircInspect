import pennylane as qp
from typing import Callable, Sequence
from pennylane.tape import QuantumTape
import numpy as np
from functools import partial

@qp.transform
def replace_paulix(tape: qp.tape.QuantumTape)-> (Sequence[QuantumTape], Callable):
    new_ops = []
    for op in tape.operations:
        if op.name == "PauliX":
            new_ops.append(qp.Hadamard(op.wires))
            new_ops.append(qp.PauliZ(op.wires))
            new_ops.append(qp.Hadamard(op.wires))
        else:
             new_ops.append(op)
    return [QuantumTape(new_ops, tape.measurements, tape.shots, tape.trainable_params)], lambda res: res[0]


@qp.transform
def replace_pauliz(tape: qp.tape.QuantumTape)-> (Sequence[QuantumTape], Callable):
    new_ops = []
    for op in tape.operations:
        if op.name == "PauliZ":
            new_ops.append(qp.S(op.wires))
            new_ops.append(qp.S(op.wires))
        else:
             new_ops.append(op)
    return [QuantumTape(new_ops, tape.measurements, tape.shots, tape.trainable_params)], lambda res: res[0]

dev = qp.device("default.qubit", wires=3)

"""@qp.qnode(dev)"""
@replace_pauliz
@replace_paulix
@qp.qnode(dev)
def my_circuit():
    qp.Hadamard(wires=0)
    return qp.state()

my_circuit()
